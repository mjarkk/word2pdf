# ICU data word2pdf does not need (patterns for grep -E, matched against `icupkg -l`); the
# bundled ICU is built without them (external/icu/ExternalProject_icu.mk).
#
# Display names of languages, regions, currencies, time zones and units: LibreOffice has its own
# locale data and only uses ICU's display names in UI code.
^lang/
^region/
^curr/
^zone/
^unit/
# Charset converter tables: LibreOffice converts text encodings itself and the bundled libxml2
# is built without ICU.
\.cnv$
# Converter aliases go with them.
^cnvalias\.icu$
# Transliteration rules: LibreOffice only uses ICU's built-in NFD, Remove and NFC
# transliterators (to ignore diacritics when searching), which need no data.
^translit/
# Line break rules: i18npool breaks lines with its own rules (OpenOffice_dat); ICU's are only
# its fallback when those do not load.
^brkitr/line
# Character names (dialogs), StringPrep profiles and the spoof checker (not linked), NFKC
# (no NFKC property is asked for). Not uts46.nrm: the PDF export converts internationalized
# domain names in links.
^unames\.icu$
\.spp$
^confusables\.cfu$
^nfkc(_cf|_scf)?\.nrm$
